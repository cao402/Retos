tabla cliente:
nombre: not null (varchar (20))
correo: primary key not null (varchar (35)/ UN)
teléfono: not null (varchar (9))

tabla ticket:
id: primary key (int automatico)
titulo: not null (varchar (15))
descripción: not null (varchar (100)
fecha y hora: not null (date/time)
fecha de cierre: null (date)
categoría: not null (el dato de opciones en "Hardware", "Software", "Redes" o "Seguridad")
prioridad: not null (el dato de opción en "Baja", "Media", "Alta" o "Critica")
Estados: not null (el dato de opción en "Abierto", "Proceso", "Cerrado" o "Archivado")

tabla operador:
nombre: not null (varchart (20))
correo: primary key not null (varchart (35)/ UN)
teléfono: not null (varchart (9))
departamentos_nivel: not null (el dato de opción en "nivel 1", "nivel 2" o "Administrador" (pequeño detalla en realidad pone etc))

tabla departamento:
nombre: primary key (varchar (20))
ubicacion: not null (varchar (20))

tabla mensaje:
id: primary key (int automatico)
texto: not null (varchar (100))
fecha y hora: not null (date/time)



1 cliente - n tickets/ 1 tickets - 1 cliente
1 departamento - n operador / 1 operador - 1 departamento
1 operador - n tickets / 1 tickets - 1 operador
1 mensaje - 1 cliente / 1 cliente - n mensajes
1 mensaje - 1 operador / 1 operador - n mensajes
1 mensaje - 1 ticket / 1 ticket - n mensajes
